CREATE PROCEDURE [dbo].[sp_PrijaviKorisnika]
    @email NVARCHAR(255),
    @lozinka NVARCHAR(255)
AS
BEGIN
    SET NOCOUNT ON;

    -- Dodat je ID na početak SELECT-a da C# ne bi bacao grešku
    SELECT ID, Email, KorisnickoIme 
    FROM Korisnik 
    WHERE Email = @email AND Lozinka = @lozinka;
END

CREATE PROCEDURE [dbo].[sp_UcitajObveznikeITransakcije]
    @Filter NVARCHAR(100) = NULL
AS
BEGIN
    SET NOCOUNT ON;

    SELECT 
        o.ID AS ObveznikID,
        o.Ime,
        o.Prezime,
        o.Email,
        o.JMBG,
        o.Opstina,
        o.Adresa,
        o.Broj,
        o.BrojTelefona,
        o.Drzava,
        o.JmbgPoreskogPunomocnika,
        t.RedniBroj AS TransakcijaID, -- Mapirano prema tvom ER dijagramu
        t.NazivAkcije,
        t.DatumPrenosa,
        t.ProdajnaCena,
        t.DatumSticanja,
        t.NabavnaCena
    FROM PoreskiObveznik o
    LEFT JOIN Transakcija t ON o.ID = t.PoreskiObveznikID
    WHERE @Filter IS NULL 
       OR o.Ime LIKE '%' + @Filter + '%'
       OR o.Prezime LIKE '%' + @Filter + '%'
       OR o.JMBG LIKE '%' + @Filter + '%';
END
